USE CASE 5: Mobile App Activity Log Analytics
1. Problem Statement

A company operates a mobile application used by thousands of customers every day.

The application generates daily activity logs containing:

User ID
User name
Event type
Device type
Event date
Session duration

The data engineering team receives these log files every day and stores them in HDFS.

The company wants to use Hive to analyze the logs and generate reports such as:

Which users generated application activity?
How many users performed each type of event?

CREATE EXTERNAL TABLE IF NOT EXISTS app_activity (

# 8. JAVA UDF
==================================================

cd ~/hive-app-logs/udf

nano UpperCaseUDF.java


# CODE

import org.apache.hadoop.hive.ql.exec.UDF;
import org.apache.hadoop.io.Text;

public class UpperCaseUDF extends UDF {

    public Text evaluate(Text input) {

        if (input == null) {
            return null;
        }

        return new Text(input.toString().toUpperCase());
    }
}


==================================================

    uppercase_name(user_name) AS user_name,
