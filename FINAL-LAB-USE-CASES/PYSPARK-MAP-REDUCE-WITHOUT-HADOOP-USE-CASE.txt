//nano employee_salary.py

from pyspark import SparkConf, SparkContext

conf = SparkConf() \
    .setAppName("EmployeeSalaryAnalysis") \
    .setMaster("local[*]") \
    .set("spark.hadoop.fs.defaultFS", "file:///")

sc = SparkContext(conf=conf)

lines = sc.textFile(
    "file:///home/ubuntu/spark-mapreduce-lab/input/employees.csv"
)

header = lines.first()

data = lines.filter(lambda line: line != header)

mapped = data.map(
    lambda line: (
        line.split(",")[2],
        (int(line.split(",")[3]), 1)
    )
)

reduced = mapped.reduceByKey(
    lambda x, y: (
        x[0] + y[0],
        x[1] + y[1]
    )
)

result = reduced.mapValues(
    lambda x: (
        x[1],
        x[0],
        x[0] / x[1]
    )
)

sorted_result = result.sortByKey()

for department, values in sorted_result.collect():
    print(
        department,
        values[0],
        values[1],
        values[2]
    )

output = sorted_result.map(
    lambda x: (
        x[0],
        x[1][0],
        x[1][1],
        x[1][2]
    )
)

output.map(
    lambda x: ",".join(map(str, x))
).coalesce(1).saveAsTextFile(
    "file:///home/ubuntu/spark-mapreduce-lab/output/department_salary"
)

sc.stop()
