# Create Mapper

nano StatusCodeMapper.py
```

```
#!/usr/bin/env python3

import sys

for line in sys.stdin:
    parts = line.strip().split()

    if len(parts) >= 4:
        status_code = parts[3]
        print(f"{status_code}\t1")
```

```
# Create Combiner

nano StatusCodeCombiner.py
```

```
#!/usr/bin/env python3

import sys

current_key = None
current_sum = 0

for line in sys.stdin:
    key, value = line.strip().split("\t")
    value = int(value)

    if current_key == key:
        current_sum += value
    else:
        if current_key is not None:
            print(f"{current_key}\t{current_sum}")

        current_key = key
        current_sum = value

if current_key is not None:
    print(f"{current_key}\t{current_sum}")
```

# Create Partitioner

nano StatusCodePartitioner.py
```

```
#!/usr/bin/env python3

import sys

for line in sys.stdin:
    key, value = line.strip().split("\t")

    status = int(key)

    if 200 <= status < 300:
        partition = 0
    else:
        partition = 1

    print(f"{partition}\t{key}\t{value}")
```

```
# Create Driver

nano StatusCodeDriver.py
```

```
#!/usr/bin/env python3

import subprocess
import sys

input_path = sys.argv[1]
output_path = sys.argv[2]

streaming_jar = "/opt/hadoop/share/hadoop/tools/lib/hadoop-streaming-3.3.6.jar"

subprocess.run([
    "hadoop",
    "jar",
    streaming_jar,
    "-input", input_path,
    "-output", output_path,
    "-mapper", "python3 StatusCodeMapper.py",
    "-combiner", "python3 StatusCodeCombiner.py",
    "-reducer", "python3 StatusCodeReducer.py",
    "-file", "StatusCodeMapper.py",
    "-file", "StatusCodeCombiner.py",
    "-file", "StatusCodeReducer.py"
], check=True)
```
