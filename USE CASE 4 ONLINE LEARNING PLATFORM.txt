------------------------------------------------
26. INNER JOIN
------------------------------------------------

Find which learners are enrolled in which courses:

SELECT
    l.learner_id,
    l.learner_name,
    c.course_id,
    c.course_name
FROM learners l
INNER JOIN enrollments e
ON l.learner_id = e.learner_id
INNER JOIN courses c
ON e.course_id = c.course_id;


------------------------------------------------
27. COUNT ENROLLMENTS BY COURSE
------------------------------------------------

SELECT
    c.course_id,
    c.course_name,
    COUNT(e.enrollment_id) AS enrollment_count
FROM courses c
LEFT JOIN enrollments e
ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name
ORDER BY enrollment_count DESC;


------------------------------------------------
28. LEFT OUTER JOIN
------------------------------------------------

Find learners who have not enrolled in any course:

SELECT
    l.learner_id,
    l.learner_name,
    e.course_id
FROM learners l
LEFT OUTER JOIN enrollments e
ON l.learner_id = e.learner_id
WHERE e.learner_id IS NULL;


------------------------------------------------
29. RIGHT OUTER JOIN
------------------------------------------------

Find enrollment records with their learner information:

SELECT
    l.learner_id,
    l.learner_name,
    e.enrollment_id,
    e.course_id
FROM learners l
RIGHT OUTER JOIN enrollments e
ON l.learner_id = e.learner_id;


------------------------------------------------
30. FULL OUTER JOIN
------------------------------------------------

Find all learners and all enrollment records:

SELECT
    l.learner_id,
    l.learner_name,
    e.enrollment_id,
    e.course_id
FROM learners l
FULL OUTER JOIN enrollments e
ON l.learner_id = e.learner_id;


------------------------------------------------
31. FIND COURSES WITH NO LEARNERS
------------------------------------------------

SELECT
    c.course_id,
    c.course_name
FROM courses c
LEFT OUTER JOIN enrollments e
ON c.course_id = e.course_id
WHERE e.course_id IS NULL;


------------------------------------------------
32. MAP-SIDE JOIN
------------------------------------------------

The courses table is small compared with a large enrollment table.

Use Hive's MAPJOIN hint:

SELECT /*+ MAPJOIN(c) */
    e.enrollment_id,
    e.learner_id,
    c.course_name
FROM enrollments e
JOIN courses c
ON e.course_id = c.course_id;


------------------------------------------------
33. REDUCE-SIDE JOIN
------------------------------------------------

Perform a normal JOIN without MAPJOIN:

SELECT
    e.enrollment_id,
    e.learner_id,
    c.course_name
FROM enrollments e
JOIN courses c
ON e.course_id = c.course_id;
